# Scaled Flash-Cross-Attention mechanism with spectral gating
import torch
import torch.nn as nn
from torch.cuda.amp import autocast

class SpectralCrossAttention(nn.Module):
    def __init__(self, dim: int = 512, heads: int = 8, tau: float = 0.07):
        super().__init__()
        self.heads = heads
        self.scale = (dim // heads) ** -0.5
        self.tau = nn.Parameter(torch.tensor(tau))
        self.to_qkv = nn.Linear(dim, dim * 3, bias=False)
        self.spectral_gate = nn.Sequential(
            nn.LayerNorm(dim),
            nn.Linear(dim, dim),
            nn.GELU()
        )

    @autocast(dtype=torch.float16)
    def forward(self, x: torch.Tensor, context: torch.Tensor = None):
        B, N, C = x.shape
        q, k, v = self.to_qkv(x).chunk(3, dim=-1)
        # Scaled Cosine similarity modulated by temperature tau
        sim = torch.einsum('b i d, b j d -> b i j', q, k) * self.scale / self.tau.clamp(min=0.01)
        attn = sim.softmax(dim=-1)
        out = torch.einsum('b i j, b j d -> b i d', attn, v)
        return self.spectral_gate(out) + x
